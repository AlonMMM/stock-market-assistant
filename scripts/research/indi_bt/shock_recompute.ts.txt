function sessions(lo: string, hi: string): string[] {
  const out: string[] = [];
  for (let d = hi; d >= lo; d = previousSessions(d, 1)[0]!) if (coreClose(d) !== null) out.unshift(d);

}

const W = fastRule.window;
/** Today's pace ratio at slot t (fastState's), 0 before the pace is measured. */
function paceAt(v: number[], cum: number[], t: number, W = fastRule.window): number {
  let win = 0;
  for (let k = t - W + 1; k <= t; k++) win += v[k] ?? 0;
  const before = t - W;
  if (before >= fastRule.paceFrom + 15 * 6 - 1 && t < fastRule.paceTo) {
    const avg = cum[before]! / (before - fastRule.paceFrom + 1);
    if (avg > 0) return win / (avg * W);
  }
  return 0;
}
type Alert = { second: number; price: number; up: boolean };
/** fastAlerts with an extra gate on each slot's state. */
/** fastWindows with a window of W slots (the live one is 6: 60 s). */
function windowsW(day: FastDay, W: number): FastWindows {
  const n = fastRule.slots, cf = carried(day.c);
  const vol: (number | null)[] = new Array(n).fill(null), move: (number | null)[] = new Array(n).fill(null);
  let sum = 0;
  for (let t = 0; t < n; t++) {
    sum += day.v[t] ?? 0;
    if (t >= W) sum -= day.v[t - W] ?? 0;
    if (t >= W - 1) vol[t] = sum;
    const now = cf[t] ?? null, before = t >= W ? (cf[t - W] ?? null) : null;
    if (now !== null && before !== null) move[t] = Math.abs((now / before - 1) * 100);
  }
  return { vol, move };
}
/** A day's 60-second volumes averaged over +-S slots around each moment (the windows that exist); the move as is. */
function smoothed(w: FastWindows, S: number): FastWindows {
  const n = w.vol.length, vol: (number | null)[] = new Array(n).fill(null);
  for (let t = 0; t < n; t++) {
    if (w.vol[t] === null) continue;
    let sum = 0, k = 0;
    for (let u = Math.max(0, t - S); u <= Math.min(n - 1, t + S); u++) if (w.vol[u] !== null) { sum += w.vol[u]!; k++; }
    vol[t] = sum / k;
  }
  return { vol, move: w.move };
}
function alertsWith(today: FastDay, base: FastBaseline, gate: (t: number, pace: number) => boolean, volume = shockRule.volume, W = fastRule.window): Alert[] {
  // A longer window scales the move and share floors as the live rule set them for 60 s (fast.ts: x sqrt(W/18), x W/18).
  const rule = { ...shockRule, volume, window: W, minMovePercent: 0.5 * Math.sqrt(W / 18), minShares: (10000 * W) / 18 };
  const cf = carried(today.c), cum = paceSums(today.v), fire = new FastFire(), out: Alert[] = [];
  for (let t = 0; t < fastRule.slots; t++) {
    const st = fastState(today, cf, base, t, cum, rule);
    const on = st.on && gate(t, paceAt(today.v, cum, t, W));
    if (!fire.step(t, on)) continue;
    if (t < shockRule.firstSlot || t > shockRule.lastSlot) continue;
    out.push({ second: (t + 1) * 10, price: cf[t]!, up: st.up });
  }
  return out;
}
